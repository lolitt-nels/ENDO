       t.Annee,
       SUM(f.DureeSejour)                                    AS DureeGlobale,
       NTILE(3) OVER (ORDER BY SUM(f.DureeSejour) DESC)      AS Segment
FROM   FSejour f
JOIN   DTemps  t ON f.CodeTemps = t.CodeTemps
GROUP BY t.Mois, t.Annee
ORDER BY t.Annee, t.Mois;

SELECT h.NomWilaya,
       s.NomSpecialite,
       SUM(f.DureeSejour)                                AS DureeGlobale,
       SUM(SUM(f.DureeSejour)) OVER (
           PARTITION BY h.NomWilaya
           ORDER BY SUM(f.DureeSejour) DESC
           ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
       )                                                  AS DureeCumul,
       ROUND(
           SUM(SUM(f.DureeSejour)) OVER (
               PARTITION BY h.NomWilaya
               ORDER BY SUM(f.DureeSejour) DESC
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) * 100 /
           SUM(SUM(f.DureeSejour)) OVER (PARTITION BY h.NomWilaya)
       , 2)                                               AS PctCumul
FROM   FSejour     f
JOIN   DHopital    h ON f.IDHopital    = h.IDHopital
JOIN   DSpecialite s ON f.IDSpecialite = s.IDSpecialite
GROUP BY h.NomWilaya, s.NomSpecialite
ORDER BY h.NomWilaya, DureeGlobale DESC;

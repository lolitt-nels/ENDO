SELECT t.Annee,
       s.NomSpecialite,
       SUM(f.NBActes)                                              AS NbActes,
       ROUND(
           SUM(f.NBActes) * 100 /
           SUM(SUM(f.NBActes)) OVER (PARTITION BY t.Annee)
       , 2)                                                         AS RatioPct
FROM   FSejour     f
JOIN   DSpecialite s ON f.IDSpecialite = s.IDSpecialite
JOIN   DTemps      t ON f.CodeTemps    = t.CodeTemps
GROUP BY t.Annee, s.NomSpecialite
ORDER BY t.Annee, NbActes DESC;

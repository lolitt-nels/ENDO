SELECT DECODE(GROUPING(h.NomWilaya),    1, 'TOUTES WILAYAS',    h.NomWilaya)    AS Wilaya,
       DECODE(GROUPING(s.NomSpecialite),1, 'TOUTES SPECIALITES', s.NomSpecialite) AS Specialite,
       DECODE(GROUPING(t.Annee),        1, 'TOUTES ANNEES',      t.Annee)        AS Annee,
       SUM(f.NBActes)                                                             AS NbActes,
       GROUPING_ID(h.NomWilaya, s.NomSpecialite, t.Annee)                        AS GID
FROM   FSejour     f
JOIN   DHopital    h ON f.IDHopital    = h.IDHopital
JOIN   DSpecialite s ON f.IDSpecialite = s.IDSpecialite
JOIN   DTemps      t ON f.CodeTemps    = t.CodeTemps
GROUP BY ROLLUP(h.NomWilaya, s.NomSpecialite, t.Annee)
ORDER BY h.NomWilaya, s.NomSpecialite, t.Annee;

SET TIMING ON;

SELECT h.NomWilaya,
       s.NomSpecialite,
       t.Annee,
       SUM(f.NBActes) AS NbActes
FROM   FSejour     f
JOIN   DHopital    h ON f.IDHopital    = h.IDHopital
JOIN   DSpecialite s ON f.IDSpecialite = s.IDSpecialite
JOIN   DTemps      t ON f.CodeTemps    = t.CodeTemps
GROUP BY CUBE(h.NomWilaya, s.NomSpecialite, t.Annee)
ORDER BY h.NomWilaya, s.NomSpecialite, t.Annee;

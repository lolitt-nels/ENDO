SELECT t.Annee,
       h.NomWilaya,
       SUM(f.NBActes)                                                        AS NbActes,
       RANK()       OVER (PARTITION BY t.Annee ORDER BY SUM(f.NBActes) DESC) AS Rang_NonDense,
       DENSE_RANK() OVER (PARTITION BY t.Annee ORDER BY SUM(f.NBActes) DESC) AS Rang_Dense
FROM   FSejour  f
JOIN   DHopital h ON f.IDHopital = h.IDHopital
JOIN   DTemps   t ON f.CodeTemps = t.CodeTemps
GROUP BY t.Annee, h.NomWilaya
ORDER BY t.Annee, NbActes DESC;

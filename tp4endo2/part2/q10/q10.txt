ALTER SYSTEM FLUSH SHARED_POOL;
ALTER SYSTEM FLUSH BUFFER_CACHE;
SET TIMING ON;

SELECT t.Annee,
       SUM(f.NBActes)                                                    AS NbActes,
       AVG(SUM(f.NBActes)) OVER (
           ORDER BY t.Annee
           ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
       )                                                                  AS MoyMobile3
FROM   FSejour  f
JOIN   DHopital h ON f.IDHopital = h.IDHopital
JOIN   DTemps   t ON f.CodeTemps = t.CodeTemps
WHERE  UPPER(h.NomWilaya) = 'ALGER'
GROUP BY t.Annee
ORDER BY t.Annee;

SET TIMING ON;

SELECT Annee, NomWilaya, NbActes
FROM (
    SELECT t.Annee,
           h.NomWilaya,
           SUM(f.NBActes)                                                        AS NbActes,
           RANK() OVER (PARTITION BY t.Annee ORDER BY SUM(f.NBActes) DESC)       AS Rang
    FROM   FSejour  f
    JOIN   DHopital h ON f.IDHopital = h.IDHopital
    JOIN   DTemps   t ON f.CodeTemps = t.CodeTemps
    GROUP BY t.Annee, h.NomWilaya
)
WHERE Rang = 1
ORDER BY Annee;
